export type ProxyLifecycleDto = "stopped" | "starting" | "running" | "draining" | "stopping" | "failed";

export type ProxyStatusDto = {
  running: boolean;
  lifecycle: ProxyLifecycleDto;
  bindAddr: string;
  port: number;
  startedAt: string | null;
  lastError: string | null;
  activeRequests: number;
  requestCount: number;
};

export type RouteHealthStateDto = "ready" | "cooldown" | "degraded" | "offline" | "unknown";
export type DecisionFactKindDto = "capability" | "health" | "pricing" | "balance" | "policy";
export type DecisionFactSeverityDto = "info" | "warning" | "error";

export type DecisionFactDto = {
  kind: DecisionFactKindDto;
  label: string;
  value: string;
  severity: DecisionFactSeverityDto;
};

export type LocalRoutingSettingsDto = {
  enabled: boolean;
  bindAddr: string;
  port: number;
  endpoint: RouteEndpointKindDto;
  policy: string;
  maxRateMultiplier: number | null;
  routingGroupFilter: RoutingGroupFilterDto;
  fallbackEnabled: boolean;
  previewKind: "baseline_eligibility";
};

export type LocalRoutingSummaryDto = {
  candidateCount: number;
  previewEligibleCandidateCount: number;
  previewExcludedCandidateCount: number;
  cooldownCandidateCount: number;
  lastDecisionAt: string | null;
};

export type LocalRoutingCandidateRowDto = {
  stationKeyId: string;
  stationId: string;
  stationName: string;
  keyName: string;
  endpoint: RouteEndpointKindDto;
  priority: number;
  enabled: boolean;
  schedulable: boolean;
  healthState: RouteHealthStateDto;
  lastSuccessAt: string | null;
  lastFailureAt: string | null;
  cooldownUntil: string | null;
  score: number | null;
  effectiveMultiplier: number | null;
  effectiveMultiplierSource: string | null;
  effectiveMultiplierConfidence: number | null;
  routingGroupScope: RoutingGroupFilterDto;
  routingGroupMatch: boolean;
  schedulerRejectReason: string | null;
  previewEligible: boolean;
  previewRejectReasons: string[];
  facts: DecisionFactDto[];
};

export type RouteDecisionSummaryDto = {
  id: string;
  decidedAt: string;
  endpoint: RouteEndpointKindDto;
  model: string | null;
  selectedStationKeyId: string | null;
  selectedStationId: string | null;
  selectedStationName: string | null;
  policy: string;
  status: "selected" | "fallback" | "failed" | "unavailable";
  reason: string;
  fallbackCount: number;
};

export type RouteDecisionEventDto = {
  id: string;
  decisionId: string;
  occurredAt: string;
  stationKeyId: string | null;
  stationId: string | null;
  accepted: boolean;
  facts: DecisionFactDto[];
  message: string;
};

export type LocalRoutingWorkspaceDto = {
  proxyStatus: ProxyStatusDto;
  settings: LocalRoutingSettingsDto;
  summary: LocalRoutingSummaryDto;
  candidates: LocalRoutingCandidateRowDto[];
  latestDecision: RouteDecisionSummaryDto | null;
  recentEvents: RouteDecisionEventDto[];
};
